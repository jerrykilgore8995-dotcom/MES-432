cmake_minimum_required(VERSION 3.22)
project(MES432 VERSION 0.1.0)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(MES432 COMPANY_NAME "MES Audio" PLUGIN_MANUFACTURER_CODE Mesa PLUGIN_CODE Me43 FORMATS VST3 PRODUCT_NAME "MES-432" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE)
juce_generate_juce_header(MES432)
target_sources(MES432 PRIVATE Source/Plugin.cpp)
target_compile_features(MES432 PRIVATE cxx_std_17)
target_compile_definitions(MES432 PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(MES432 PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)

juce_add_console_app(MESMeasure PRODUCT_NAME "MESMeasure")
target_sources(MESMeasure PRIVATE Tools/Measure.cpp)
target_compile_features(MESMeasure PRIVATE cxx_std_17)
target_compile_definitions(MESMeasure PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(MESMeasure PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)

juce_add_console_app(MESCapture PRODUCT_NAME "MESCapture")
target_sources(MESCapture PRIVATE Tools/Capture.cpp)
target_compile_features(MESCapture PRIVATE cxx_std_17)
target_compile_definitions(MESCapture PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(MESCapture PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
